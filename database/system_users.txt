SELECT 
    u.email,
    p.full_name,
    ur.role,
    a.agency_name,
    substring(p.agency_id::text, 1, 8) as agency_id_prefix
FROM auth.users u
JOIN public.profiles p ON u.id = p.id
JOIN public.user_roles ur ON u.id = ur.user_id
JOIN public.agency a ON p.agency_id = a.id
WHERE u.email IN ('admin@test.com', 'munkh.mn@gmail.com')
ORDER BY ur.role DESC;
